\begin{minipage}[t]{0.49\textwidth}\centering
\textbf{(a) Task 1: universal autoencoder}\\[1mm]
\begin{tikzpicture}[node distance=4.5mm]

\node[blk, right=of x] (e) {encoder\\4 stride-2\\stages};

\node[blk, right=of z] (d) {decoder\\4 up-\\sampling\\stages};

\begin{minipage}[t]{0.49\textwidth}\centering

\begin{tikzpicture}[node distance=4mm]

\begin{minipage}[t]{0.49\textwidth}\centering

\caption{Architectures of the four systems. (a) Task~1: a strict-bottleneck convolutional autoencoder without skip connections. (b) Task~2: a classifier selects exactly one branch (the identity for clean images, otherwise one of three specialist autoencoders). (c) Task~3: a gate assigns softmax weights to the identity branch and the three experts, and the output is the weighted sum. (d) Task~4: a U-Net generator conditioned on a learned style embedding (concatenation and FiLM) and a PatchGAN discriminator that also sees the embedding.}
\label{fig:arch}
\end{figure*}
